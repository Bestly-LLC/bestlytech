# Minutes are jittered off :00/:30 so they do not pile onto other jobs. Logs go where the other jobs' logs go.
# worker: every 2 hours at :17
17 */2 * * * /opt/bestly/cron/run.sh skill_inbox >> /home/pi/scripts/cron-logs/skill_inbox.log 2>&1

# watchdog: every 3 hours at :41
41 */3 * * * /opt/bestly/cron/run.sh skill_inbox_watch >> /home/pi/scripts/cron-logs/skill_inbox_watch.log 2>&1
